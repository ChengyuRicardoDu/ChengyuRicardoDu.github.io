\section{PROFILE}
\begin{flushleft}
\begin{adjustwidth}{0pt}{0pt}
\justifying
\small
I am an interdisciplinary Ph.D. candidate working at the intersection of
\textbf{statistics}, \textbf{machine learning}, and \textbf{computational biology}.
My research develops statistical and computational methods for structured,
high-dimensional biological data, including
\textbf{spatial transcriptomics},
\textbf{3D genome organization (Hi-C)},
and \textbf{RNA structure ensembles}.
Methodologically, I am interested in
\textbf{kernel methods},
\textbf{representation learning},
\textbf{high-dimensional inference},
\textbf{scan statistics},
and \textbf{optimization-based approaches}
for recovering biological structure from sparse and noisy measurements.
\end{adjustwidth}
\end{flushleft}
